function multiply() {
    let num1 = document.getElementById("num1");
    let div = document.getElementById("div");

    div.innerHTML = "계산 중...";

    let addWorker = new Worker("worker.js");     // 워커 생성

    addWorker.onmessage = function (e) {         // 워커가 보낸 결과 받기
        div.innerHTML = e.data.join("<br>");
        addWorker.terminate();
    };

    addWorker.postMessage(num1.value);           // 워커에 숫자 전달
}

worker.js

onmessage = function (e) {
    let dan = Number(e.data);   // 메인에서 받은 숫자
    let result = [];

    for (let i = 1; i <= 9; i++) {
        result.push(dan + " x " + i + " = " + (dan * i));
    }

    // "계산 중..."이 보이도록 1초 뒤에 결과 전달
    setTimeout(function () {
        postMessage(result);    // 결과를 메인 스레드로 보냄
    }, 1000);
};
